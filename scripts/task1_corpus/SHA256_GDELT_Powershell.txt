
$currentFolder = (Get-Location).Path
$baseFolder = $null

while ($currentFolder) {

    $candidate = Join-Path $currentFolder "gdelt_organised"

    if (Test-Path $candidate -PathType Container) {
        $baseFolder = $candidate
        break
    }

    if ((Split-Path $currentFolder -Leaf) -eq "gdelt_organised") {
        $baseFolder = $currentFolder
        break
    }

    $parentFolder = Split-Path $currentFolder -Parent

    if ($parentFolder -eq $currentFolder) {
        break
    }

    $currentFolder = $parentFolder
}

if (!$baseFolder) {
    Write-Error "Cannot find gdelt_organised folder."
    return
}

Write-Output "Found dataset folder: $baseFolder"

$years = @("2015", "2016", "2017")

foreach ($year in $years) {

    $folder = Join-Path $baseFolder $year

    if (!(Test-Path $folder -PathType Container)) {
        Write-Warning "Folder not found: $year"
        continue
    }

    $records = @(Get-ChildItem $folder -File -Recurse -Filter "*.CSV" |
        ForEach-Object {
            $relativePath = $_.FullName.Substring($folder.Length).TrimStart('\').Replace('\', '/')
            $hash = (Get-FileHash $_.FullName -Algorithm SHA256).Hash.ToLowerInvariant()
            "$relativePath $hash"
        })

    [Array]::Sort($records, [System.StringComparer]::Ordinal)

    $combined = if ($records.Count -gt 0) {
        ($records -join "`n") + "`n"
    } else {
        ""
    }

    $bytes = [System.Text.Encoding]::UTF8.GetBytes($combined)

    $sha256 = [System.Security.Cryptography.SHA256]::Create()

    try {
        $hashBytes = $sha256.ComputeHash($bytes)
        $folderHash = [System.BitConverter]::ToString($hashBytes).Replace("-", "").ToLowerInvariant()

        Write-Output "$year Folder SHA256: $folderHash"
    }
    finally {
        $sha256.Dispose()
    }
}
